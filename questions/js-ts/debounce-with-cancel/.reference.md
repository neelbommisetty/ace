# Reference Solution

````tsx
export interface Debounced<TArgs extends unknown[]> {
  (...args: TArgs): void;
  cancel(): void;
  flush(): void;
  pending(): boolean;
}

export function debounce<TArgs extends unknown[]>(
  fn: (...args: TArgs) => void,
  waitMs: number,
): Debounced<TArgs> {
  const delay = Math.max(0, waitMs);
  let timer: ReturnType<typeof setTimeout> | null = null;
  let queued: TArgs | null = null;

  // Clears the schedule BEFORE calling fn, so pending() reads false from
  // inside fn and a re-entrant call starts a clean window.
  function invoke(): void {
    const args = queued;
    timer = null;
    queued = null;
    if (args !== null) fn(...args);
  }

  const debounced = Object.assign(
    (...args: TArgs): void => {
      queued = args;
      if (timer !== null) clearTimeout(timer);
      timer = setTimeout(invoke, delay);
    },
    {
      cancel(): void {
        if (timer !== null) clearTimeout(timer);
        timer = null;
        queued = null;
      },
      flush(): void {
        if (timer === null) return;
        clearTimeout(timer);
        invoke();
      },
      pending(): boolean {
        return timer !== null;
      },
    },
  );

  return debounced;
}
````
