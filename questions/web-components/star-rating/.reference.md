# Reference Solution

````tsx
import type { KeyboardEvent, ReactElement } from 'react';

export interface StarRatingProps {
  value: number;
  onChange: (next: number) => void;
  max?: number;
  label: string;
  readOnly?: boolean;
}

export function StarRating({
  value,
  onChange,
  max = 5,
  label,
  readOnly = false,
}: StarRatingProps): ReactElement {
  const stars = Array.from({ length: max }, (_, index) => index + 1);

  // Single funnel for every input path: clamp once, honour readOnly once.
  const emit = (next: number): void => {
    if (readOnly) return;
    onChange(Math.min(max, Math.max(0, next)));
  };

  const handleKeyDown = (event: KeyboardEvent<HTMLDivElement>): void => {
    switch (event.key) {
      case 'ArrowRight':
      case 'ArrowUp':
        emit(value + 1);
        break;
      case 'ArrowLeft':
      case 'ArrowDown':
        emit(value - 1);
        break;
      case 'Home':
        emit(1);
        break;
      case 'End':
        emit(max);
        break;
      default:
        return; // unrelated key: leave it to the browser
    }
    event.preventDefault();
  };

  return (
    <div
      role="radiogroup"
      aria-label={label}
      aria-readonly={readOnly || undefined}
      tabIndex={0}
      onKeyDown={handleKeyDown}
    >
      {stars.map((star) => (
        <button
          key={star}
          type="button"
          role="radio"
          aria-checked={star === value}
          aria-label={`${star} star${star === 1 ? '' : 's'}`}
          disabled={readOnly}
          // Clicking the current rating clears it; anything else selects it.
          onClick={() => emit(star === value ? 0 : star)}
        >
          {star <= value ? '★' : '☆'}
        </button>
      ))}
    </div>
  );
}
````
