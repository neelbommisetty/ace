# Reference Solution

````tsx
/**
 * Kadane's algorithm. `bestEndingHere` is the largest sum of a window that
 * ends at index i: either the previous window extended by deltas[i], or a
 * brand-new window starting at deltas[i]. Seeding both accumulators with
 * deltas[0] (rather than 0) is what makes the all-negative case return the
 * least-negative element instead of 0.
 */
export function bestRevenueWindow(deltas: number[]): number {
  if (deltas.length === 0) return 0;

  let bestEndingHere = deltas[0];
  let best = deltas[0];

  for (let i = 1; i < deltas.length; i++) {
    bestEndingHere = Math.max(deltas[i], bestEndingHere + deltas[i]);
    best = Math.max(best, bestEndingHere);
  }

  return best;
}
````
