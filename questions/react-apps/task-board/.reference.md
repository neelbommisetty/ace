# Reference Solution

````tsx
import { useState, type FormEvent, type ReactElement } from 'react';

export interface Task {
  id: string;
  title: string;
  done: boolean;
}

export type Filter = 'all' | 'active' | 'done';

const FILTERS: Array<{ key: Filter; label: string }> = [
  { key: 'all', label: 'All' },
  { key: 'active', label: 'Active' },
  { key: 'done', label: 'Done' },
];

let nextId = 0;

export default function App(): ReactElement {
  const [tasks, setTasks] = useState<Task[]>([]);
  const [draft, setDraft] = useState('');
  const [filter, setFilter] = useState<Filter>('all');

  // Derived at render time — never stored, so the counter below cannot drift
  // out of sync with the filter.
  const visible = tasks.filter((task) =>
    filter === 'all' ? true : filter === 'done' ? task.done : !task.done,
  );
  const doneCount = tasks.filter((task) => task.done).length;

  function handleSubmit(event: FormEvent<HTMLFormElement>): void {
    event.preventDefault();
    const title = draft.trim();
    if (title === '') return; // blank submit: keep the draft, add nothing
    setTasks((prev) => [...prev, { id: `task-${nextId++}`, title, done: false }]);
    setDraft('');
  }

  function toggle(id: string): void {
    setTasks((prev) =>
      prev.map((task) => (task.id === id ? { ...task, done: !task.done } : task)),
    );
  }

  return (
    <main>
      <form onSubmit={handleSubmit}>
        <label htmlFor="new-task">New task</label>
        <input
          id="new-task"
          type="text"
          value={draft}
          onChange={(event) => setDraft(event.target.value)}
        />
        <button type="submit">Add task</button>
      </form>

      <div>
        {FILTERS.map(({ key, label }) => (
          <button
            key={key}
            type="button"
            aria-pressed={filter === key}
            onClick={() => setFilter(key)}
          >
            {label}
          </button>
        ))}
        <span>
          {doneCount} of {tasks.length} done
        </span>
      </div>

      {visible.length === 0 ? (
        <p>Nothing here yet.</p>
      ) : (
        <ul>
          {visible.map((task) => (
            <li key={task.id}>
              <label>
                <input type="checkbox" checked={task.done} onChange={() => toggle(task.id)} />
                {task.title}
              </label>
            </li>
          ))}
        </ul>
      )}

      <button
        type="button"
        disabled={doneCount === 0}
        onClick={() => setTasks((prev) => prev.filter((task) => !task.done))}
      >
        Clear done
      </button>
    </main>
  );
}
````
